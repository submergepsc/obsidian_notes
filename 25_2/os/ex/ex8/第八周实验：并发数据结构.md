# 第八周实验：并发数据结构

# 朴素并发链表

- 直接使用 pthread 库实现并发形式的链表插入操作

    - 每个线程把若干节点头插到同一个全局链表 g\_head 上

```C
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

typedef struct node {
    int          key;
    struct node *next;
} node_t;

/* 全局共享链表头 */
static node_t *g_head = NULL;

typedef struct {
    int tid;
    int begin;   /* inclusive */
    int end;     /* exclusive */
} worker_t;

/* "朴素"的头插: 读 g_head -> 写 n->next -> 写 g_head。
 * 在单线程里它完全正确; 多线程并发调用时可能会出什么事 */
static void insert_front(int key)
{
    node_t *n = (node_t *)malloc(sizeof(*n));
    if (!n) { perror("malloc"); exit(1); }
    n->key  = key;
    n->next = g_head;   /* (A) 读共享状态 */
    g_head  = n;        /* (B) 写共享状态 */
}

static void *worker_fn(void *arg)
{
    worker_t *w = (worker_t *)arg;
    for (int k = w->begin; k < w->end; ++k) {
        insert_front(k);
    }
    return NULL;
}

static size_t list_length(const node_t *head)
{
    size_t n = 0;
    for (const node_t *p = head; p; p = p->next) ++n;
    return n;
}

static void free_list(node_t *head)
{
    while (head) {
        node_t *nx = head->next;
        free(head);
        head = nx;
    }
}

int main(int argc, char **argv)
{
    int nthreads = (argc > 1) ? atoi(argv[1]) : 8;
    int per      = (argc > 2) ? atoi(argv[2]) : 100000;
    if (nthreads <= 0 || per <= 0) {
        fprintf(stderr, "usage: %s [num_threads] [inserts_per_thread]\n", argv[0]);
        return 1;
    }

    pthread_t *tids = calloc(nthreads, sizeof(*tids));
    worker_t  *ws   = calloc(nthreads, sizeof(*ws));
    if (!tids || !ws) { perror("calloc"); return 1; }

    struct timespec t0, t1;
    clock_gettime(CLOCK_MONOTONIC, &t0);

    for (int i = 0; i < nthreads; ++i) {
        ws[i].tid   = i;
        ws[i].begin = i * per;
        ws[i].end   = (i + 1) * per;
        if (pthread_create(&tids[i], NULL, worker_fn, &ws[i]) != 0) {
            perror("pthread_create");
            return 1;
        }
    }
    for (int i = 0; i < nthreads; ++i) pthread_join(tids[i], NULL);

    clock_gettime(CLOCK_MONOTONIC, &t1);
    double ms = (t1.tv_sec - t0.tv_sec) * 1e3 +
                (t1.tv_nsec - t0.tv_nsec) / 1e6;

    size_t got      = list_length(g_head);
    size_t expected = (size_t)nthreads * (size_t)per;
    long   lost     = (long)expected - (long)got;

    printf("threads=%d  per-thread=%d  elapsed=%.2f ms\n",
           nthreads, per, ms);
    printf("merged list length = %zu (expected %zu, lost %ld) : %s\n",
           got, expected, lost, got == expected ? "OK" : "MISMATCH");

    free_list(g_head);
    free(ws);
    free(tids);
    return got == expected ? 0 : 2;  
}

```

# 创建线程私有子链表

## 要求

在不使用 pthread\_mutex / 不使用 atomic 的前提下，让多个线程\&\#34;并发地\&\#34;完成链表插入，并在主线程合并为一条全局链表。

## 设计约束

1. 整个程序不得包含 pthread\_mutex\_\* / sem\_\* / \&lt;stdatomic\.h\&gt; / \_\_sync\_\* / \_\_atomic\_\* / volatile 等任何同步或原子原语。

2. 必须通过 pthread\_create / pthread\_join 完成并发。

3. 主线程最后输出的全局链表长度必须等于 nthreads \* inserts\_per\_thread。

## 提示

1. 并发阶段彼此不写同一块内存

2. 合并阶段利用 pthread\_join 在主线程内完成

3. 不要让多个线程去改同一个 head 指针
